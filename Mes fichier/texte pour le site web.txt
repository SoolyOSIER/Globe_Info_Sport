<script>
  // Header background + scroll progress
  const header = document.getElementById('siteHeader');
  const progressBar = document.getElementById('progressBar');
  window.addEventListener('scroll', () => {
    header.classList.toggle('scrolled', window.scrollY > 40);
    const h = document.documentElement;
    const pct = (h.scrollTop) / (h.scrollHeight - h.clientHeight) * 100;
    progressBar.style.width = pct + '%';
  });

  // Mobile nav toggle
  const navToggle = document.getElementById('navToggle');
  const navLinks = document.getElementById('navLinks');
  navToggle.addEventListener('click', () => navLinks.classList.toggle('open'));
  navLinks.querySelectorAll('a').forEach(a => a.addEventListener('click', () => navLinks.classList.remove('open')));

  // Scroll-spy active nav link
  const sections = document.querySelectorAll('section[id]');
  const spyLinks = navLinks.querySelectorAll('a');
  const spy = new IntersectionObserver((entries)=>{
    entries.forEach(e=>{
      if(e.isIntersecting){
        spyLinks.forEach(l=>l.classList.remove('active'));
        const link = navLinks.querySelector(`a[href="#${e.target.id}"]`);
        if(link) link.classList.add('active');
      }
    });
  }, {rootMargin:"-45% 0px -45% 0px"});
  sections.forEach(s=>spy.observe(s));

  // Scroll reveal (single + staggered groups)
  const revealEls = document.querySelectorAll('.reveal');
  const io = new IntersectionObserver((entries) => {
    entries.forEach(e => { if(e.isIntersecting){ e.target.classList.add('in'); io.unobserve(e.target); } });
  }, { threshold: 0.15 });
  revealEls.forEach(el => io.observe(el));

  const staggerEls = document.querySelectorAll('.stagger');
  const staggerIO = new IntersectionObserver((entries) => {
    entries.forEach(e => {
      if(e.isIntersecting){
        [...e.target.children].forEach((child,i)=>{
          child.style.transitionDelay = (i*80)+'ms';
        });
        e.target.classList.add('in');
        staggerIO.unobserve(e.target);
      }
    });
  }, { threshold: 0.1 });
  staggerEls.forEach(el => staggerIO.observe(el));

  // Animated counters
  const counters = document.querySelectorAll('.stat .num');
  const countIO = new IntersectionObserver((entries)=>{
    entries.forEach(e=>{
      if(e.isIntersecting){
        const el = e.target;
        const target = parseInt(el.dataset.count, 10);
        const isYear = el.dataset.format === 'year';
        const duration = 1400;
        const start = performance.now();
        function tick(now){
          const p = Math.min((now-start)/duration, 1);
          const eased = 1 - Math.pow(1-p, 3);
          const val = Math.floor(eased * target);
          el.textContent = isYear ? val : val;
          if(p<1) requestAnimationFrame(tick); else el.textContent = target;
        }
        requestAnimationFrame(tick);
        countIO.unobserve(el);
      }
    });
  }, {threshold:0.5});
  counters.forEach(c=>countIO.observe(c));

  // Generate hero rays
  const raysContainer = document.getElementById('rays');
  const rayCount = 7;
  for(let i=0;i<rayCount;i++){
    const ray = document.createElement('div');
    ray.className = 'ray';
    const angle = -30 + (60/(rayCount-1))*i;
    ray.style.transform = `translateX(-50%) rotate(${angle}deg)`;
    ray.style.opacity = 0.5 - Math.abs(angle)/90;
    raysContainer.appendChild(ray);
  }

  // Generate stars
  const starsContainer = document.getElementById('stars');
  for(let i=0;i<50;i++){
    const s = document.createElement('div');
    s.className = 'star';
    s.style.left = Math.random()*100+'%';
    s.style.top = Math.random()*60+'%';
    s.style.animationDelay = (Math.random()*3.5)+'s';
    starsContainer.appendChild(s);
  }

  // Cursor parallax on hero glow
  const heroGlow = document.getElementById('heroGlow');
  document.querySelector('.hero').addEventListener('mousemove', (e)=>{
    const x = (e.clientX / window.innerWidth - 0.5) * 40;
    heroGlow.style.marginLeft = x + 'px';
  });
</script>
